final results = <String, Object>{};
Finder by(String value) => find.byKey(ValueKey(value));

Future<void> capture(WidgetTester tester, String name, Map<String, Finder> items) async {
  await tester.pumpAndSettle();
  results[name] = {
    for (final item in items.entries)
      if (item.value.evaluate().length == 1)
        item.key: (() {
          final rect = tester.getRect(item.value);
          return [rect.left, rect.top, rect.width, rect.height];
        })(),
    'safeArea': [tester.view.padding.top, tester.view.padding.bottom],
    'hitAtCenter': {
      for (final item in items.entries)
        if (item.value.evaluate().length == 1) item.key: item.value.hitTestable().evaluate().length == 1,
    },
  };
  final boundary = shot.currentContext!.findRenderObject()! as RenderRepaintBoundary;
  await tester.runAsync(() async {
    final image = await boundary.toImage(pixelRatio: 1);
    final bytes = (await image.toByteData(format: ui.ImageByteFormat.png))!;
    File('$out/flutter-$name.png').writeAsBytesSync(bytes.buffer.asUint8List());
    image.dispose();
    File('$out/flutter.json').writeAsStringSync(const JsonEncoder.withIndent('  ').convert(results));
  });
}

Future<void> loadFace(String family, List<String> paths) async {
  final loader = FontLoader(family);
  for (final path in paths) {
    loader.addFont(File(path).readAsBytes().then(ByteData.sublistView));
  }
  await loader.load();
}

Future<void> loadCaptureFonts() async {
  await loadFace(primaryFont, [for (final weight in [400, 500, 600, 700]) '$fontRoot/SFNS-$weight.ttf']);
  await loadFace(chineseFont, [for (final weight in [400, 500, 600]) '$fontRoot/PingFangSC-$weight.otf']);
  await loadFace('MaterialIcons', [
    '/Users/yangyitian/flutter-3.47.9/bin/cache/artifacts/material_fonts/MaterialIcons-Regular.otf',
  ]);
  await loadFace('ProbeNerd', [
    'assets/shared/fonts/JetBrainsMonoNerdFontMono-Regular.ttf',
  ]);
}

Future<void> mount(WidgetTester tester, HostCatalog catalog) async {
  tester.view.devicePixelRatio = 1;
  tester.view.physicalSize = Platform.environment['PROBE_WIDTH'] == '320' ? const Size(320, 640) : const Size(390, 844);
  tester.view.padding = safePadding;
  tester.view.viewPadding = safePadding;
  tester.platformDispatcher.localeTestValue = const Locale('zh', 'CN');
  final tokens = (await tester.runAsync(() async {
    await (fontsReady ??= loadCaptureFonts());
    return DesignTokens.load();
  }))!;
  final theme = tokens.theme;
  final captureTheme = theme.copyWith(
    platform: TargetPlatform.iOS,
    textTheme: theme.textTheme.apply(fontFamily: primaryFont, fontFamilyFallback: fallbackFonts),
    primaryTextTheme: theme.primaryTextTheme.apply(fontFamily: primaryFont, fontFamilyFallback: fallbackFonts),
    chipTheme: theme.chipTheme.copyWith(labelStyle: theme.chipTheme.labelStyle?.copyWith(
      fontFamily: primaryFont, fontFamilyFallback: fallbackFonts)),
    appBarTheme: theme.appBarTheme.copyWith(titleTextStyle: theme.appBarTheme.titleTextStyle?.copyWith(
      fontFamily: primaryFont, fontFamilyFallback: fallbackFonts)),
  );
  await tester.pumpWidget(RepaintBoundary(key: shot, child: MaterialApp(debugShowCheckedModeBanner:false,
    theme: captureTheme, builder: (context, child) => MediaQuery(data: MediaQuery.of(context).copyWith(textScaler: TextScaler.linear(Platform.environment['PROBE_SCALE'] == '200' ? 2 : 1)), child: child!), home: HostCatalogPage(tokens: tokens, catalog: catalog, offlineBuilder: (_) => const SizedBox()))));
  await tester.pumpAndSettle();
}
